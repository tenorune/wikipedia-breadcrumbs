# PWA Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Progressive Web App for viewing and managing Wikipedia breadcrumb trails, with offline support and anonymous Supabase sync.

**Architecture:** SvelteKit static SPA (`adapter-static`, no SSR) in `packages/pwa/`. Uses `@wikipedia-breadcrumbs/shared` for IndexedDB data access and sync. Bottom tab bar navigation. Mobile-first layout. Auth is deferred to a separate plan — this PWA uses anonymous sync only.

**Tech Stack:** SvelteKit, Svelte 5 (runes), adapter-static, `@wikipedia-breadcrumbs/shared`, `@supabase/supabase-js`, Vite

**Spec reference:** `docs/superpowers/specs/2026-03-31-pwa-auth-design.md`

**Prerequisite:** `.env` file with `PUBLIC_SUPABASE_URL` and `PUBLIC_SUPABASE_ANON_KEY` (same Supabase project as the extension). Add these vars to `.env` alongside the existing `VITE_SUPABASE_*` vars.

```typescript
// packages/pwa/src/lib/stores/sync.svelte.ts
import {
  SyncEngine,
  SupabaseBackend,
  SyncStatus,
  trailStore,
  visitStore,
} from "@wikipedia-breadcrumbs/shared";
import type { SyncStateStore, SyncReport } from "@wikipedia-breadcrumbs/shared";
import { supabase } from "$lib/supabase";
import { db } from "$lib/stores/db";

let engine: SyncEngine | null = null;
let userId: string | null = null;

// Sync state — reactive via direct reads
let _syncEnabled = $state(false);
let _lastSyncTime = $state<string | null>(localStorage.getItem("lastSyncTime"));
let _syncing = $state(false);
let _lastReport = $state<SyncReport | null>(null);

export const syncState = {
  get syncEnabled() { return _syncEnabled; },
  get lastSyncTime() { return _lastSyncTime; },
  get syncing() { return _syncing; },
  get lastReport() { return _lastReport; },
};

const stateStore: SyncStateStore = {
  async getLastSyncTime() {
    return localStorage.getItem("lastSyncTime");
  },
  async setLastSyncTime(time: string) {
    localStorage.setItem("lastSyncTime", time);
    _lastSyncTime = time;
  },
};

export async function enableSync(): Promise<void> {
  if (engine) return;

  // Anonymous auth
  const { data: sessionData } = await supabase.auth.getSession();
  if (sessionData?.session?.user) {
    userId = sessionData.session.user.id;
  } else {
    const { data, error } = await supabase.auth.signInAnonymously();
    if (error || !data.user) {
      console.error("[pwa] Anonymous auth failed:", error);
      return;
    }
    userId = data.user.id;
  }

  // Stamp local trails with userId
  const allTrails = await db.trails.filter((t) => t.userId !== userId).toArray();
  for (const trail of allTrails) {
    await db.trails.update(trail.id, { userId, syncStatus: SyncStatus.PendingSync });
  }
  const allVisits = await db.visits.filter((v) => v.syncStatus !== SyncStatus.Synced).toArray();
  for (const visit of allVisits) {
    await db.visits.update(visit.id, { syncStatus: SyncStatus.PendingSync });
  }

  const backend = new SupabaseBackend(supabase, userId);
  engine = new SyncEngine(db, backend, stateStore, userId);
  _syncEnabled = true;
  localStorage.setItem("syncEnabled", "true");
}

export function disableSync(): void {
  engine = null;
  _syncEnabled = false;
  localStorage.removeItem("syncEnabled");
}

export async function syncNow(): Promise<void> {
  if (!engine) return;
  _syncing = true;
  try {
    const report = await engine.syncNow();
    _lastReport = report;
    console.log("[pwa] Sync complete:", report);
  } catch (err) {
    console.error("[pwa] Sync error:", err);
  } finally {
    _syncing = false;
  }
}

// Restore sync state on load
export async function initSync(): Promise<void> {
  if (localStorage.getItem("syncEnabled") === "true") {
    await enableSync();
    await syncNow();
    // Periodic sync every 5 minutes while tab is visible
    setInterval(() => {
      if (document.visibilityState === "visible" && engine) {
        syncNow();
      }
    }, 5 * 60 * 1000);
  }
}
```

<!-- packages/pwa/src/lib/components/TrailList.svelte -->
<script lang="ts">
  import { trailStore, visitStore } from "@wikipedia-breadcrumbs/shared";
  import type { Trail } from "@wikipedia-breadcrumbs/shared";
  import { db } from "$lib/stores/db";
  import { goto } from "$app/navigation";

  interface TrailSummary {
    trail: Trail;
    visitCount: number;
    firstTitle: string;
    lastTitle: string;
  }

  let summaries: TrailSummary[] = $state([]);
  let searchQuery = $state("");
  let sortBy: "recent" | "oldest" | "starred" = $state("recent");
  let loading = $state(true);

  const trailOps = trailStore(db);
  const visitOps = visitStore(db);

  async function refresh() {
    loading = true;
    const allTrails = await trailOps.getAll();
    const result: TrailSummary[] = [];
    for (const trail of allTrails) {
      const visits = await visitOps.getByTrailId(trail.id);
      result.push({
        trail,
        visitCount: visits.length,
        firstTitle: visits[0]?.title ?? "",
        lastTitle: visits[visits.length - 1]?.title ?? "",
      });
    }
    summaries = result;
    loading = false;
  }

  const filtered = $derived.by(() => {
    let result = summaries;
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      result = result.filter((s) =>
        s.trail.name?.toLowerCase().includes(q) ||
        s.firstTitle.toLowerCase().includes(q) ||
        s.lastTitle.toLowerCase().includes(q)
      );
    }
    if (sortBy === "recent") {
      result = [...result].sort((a, b) => b.trail.updatedAt.localeCompare(a.trail.updatedAt));
    } else if (sortBy === "oldest") {
      result = [...result].sort((a, b) => a.trail.startedAt.localeCompare(b.trail.startedAt));
    } else if (sortBy === "starred") {
      result = [...result].sort((a, b) => (b.trail.isStarred ? 1 : 0) - (a.trail.isStarred ? 1 : 0));
    }
    return result;
  });

  async function toggleStar(id: string) {
    const s = summaries.find((s) => s.trail.id === id);
    if (!s) return;
    s.trail.isStarred = !s.trail.isStarred;
    await trailOps.update(id, { isStarred: s.trail.isStarred });
    summaries = [...summaries];
  }

  async function deleteTrail(id: string) {
    await trailOps.softDelete(id);
    await refresh();
  }

  function formatDate(iso: string): string {
    return new Date(iso).toLocaleDateString();
  }

  refresh();
</script>

<div class="toolbar">
  <input type="text" placeholder="Search trails..." bind:value={searchQuery} />
  <select bind:value={sortBy}>
    <option value="recent">Most Recent</option>
    <option value="oldest">Oldest First</option>
    <option value="starred">Starred First</option>
  </select>
</div>

{#if loading}
  <p class="status">Loading trails...</p>
{:else if filtered.length === 0}
  <p class="status">{searchQuery ? "No trails match your search." : "No trails yet."}</p>
{:else}

    {#each filtered as summary}

{/if}

SvelteKit provides `$service-worker` module with build info:

```typescript
/// <reference types="@sveltejs/kit" />
/// <reference no-default-lib="true"/>
/// <reference lib="esnext" />
/// <reference lib="webworker" />

import { build, files, version } from "$service-worker";

const CACHE_NAME = `cache-${version}`;
const ASSETS = [...build, ...files];

// Install — precache app shell
self.addEventListener("install", (event: ExtendableEvent) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS))
  );
});

// Activate — clean old caches
self.addEventListener("activate", (event: ExtendableEvent) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(keys.filter((k) => k !== CACHE_NAME).map((k) => caches.delete(k)))
    )
  );
});

// Fetch — cache-first for assets, network-first for navigation
self.addEventListener("fetch", (event: FetchEvent) => {
  if (event.request.method !== "GET") return;

  const url = new URL(event.request.url);

  // Skip Supabase API calls — data is in IndexedDB
  if (url.hostname.includes("supabase")) return;

  event.respondWith(
    caches.match(event.request).then((cached) => {
      if (cached) return cached;
      return fetch(event.request).then((response) => {
        // Cache successful responses for static assets
        if (response.ok && url.origin === self.location.origin) {
          const clone = response.clone();
          caches.open(CACHE_NAME).then((cache) => cache.put(event.request, clone));
        }
        return response;
      });
    }).catch(() => {
      // Offline fallback for navigation requests
      if (event.request.mode === "navigate") {
        return caches.match("/index.html") as Promise<Response>;
      }
      return new Response("Offline", { status: 503 });
    })
  );
});
```
